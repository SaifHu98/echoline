# === Usage ===
# After installing @icons plugin, the scene tree will show these icons
# next to each node based on its name/type. This dramatically improves
# scene readability for large projects like ECHO//LINE.

# Example scene snippet:
#
# [node name="ClocktowerInstance" parent="." instance=ExtResource("clocktower.tscn")]
# → Will show clocktower.svg icon in scene tree

# [node name="PlayerAvatar" parent="."]
# → Will show player.svg icon

# [node name="PastGarden" parent="."]
# → Will show past_building.svg icon

# This makes debugging MUCH faster — you can spot the right node
# immediately instead of reading names.
